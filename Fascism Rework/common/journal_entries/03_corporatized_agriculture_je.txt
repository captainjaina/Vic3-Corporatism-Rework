je_corporatized_agriculture = {
	icon = "gfx/interface/icons/goods_icons/grain.dds"
	group = je_group_internal_affairs

	scripted_button = establish_state_grain_boards_button
	scripted_button = agrarian_mechanization_subsidies_button

	is_shown_when_inactive = {
		NOT = { has_variable = corporatized_agriculture_completed }
		active_law:lawgroup_economic_system ?= {
			has_amendment = amendment_type:amendment_corporatized_economy
		}
	}

	activation_description = {
		active_law:lawgroup_economic_system ?= {
			has_amendment = amendment_type:amendment_corporatized_economy
		}
		market ?= {
			mg:grain ?= {
				market_goods_pricier >= 0.20
			}
		}
	}

	possible = {
		active_law:lawgroup_economic_system ?= {
			has_amendment = amendment_type:amendment_corporatized_economy
		}
		market ?= {
			mg:grain ?= {
				market_goods_pricier >= 0.20
			}
		}
	}

	immediate = {
		set_variable = {
			name = corporatized_agriculture_months
			value = 0
		}
		trigger_event = {
			id = corporatized_economy.11
			popup = yes
		}
	}

	on_monthly_pulse = {
		effect = {
			change_variable = {
				name = corporatized_agriculture_months
				add = 1
			}
		}
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			market ?= {
				mg:grain ?= {
					market_goods_pricier <= 0.05
				}
			}
			OR = {
				has_law = law_type:law_homesteading
				has_law = law_type:law_commercialized_agriculture
				has_law = law_type:law_collectivized_agriculture
			}
		}
		effect = {
			random_interest_group = {
				limit = { is_in_government = yes }
				save_scope_as = agrarian_sponsor_ig
			}
			active_law:lawgroup_land_reform ?= {
				add_amendment = {
					type = amendment_type:amendment_corporatized_agriculture
					sponsor = scope:agrarian_sponsor_ig
				}
			}
			add_loyalists = {
				value = 0.15
				pop_type = farmers
			}
			set_variable = corporatized_agriculture_completed
			remove_variable = corporatized_agriculture_months
			if = {
				limit = { has_variable = agrarian_subsidies_days_left }
				remove_variable = agrarian_subsidies_days_left
			}
			trigger_event = {
				id = corporatized_economy.12
				popup = yes
			}
		}
	}

	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			NOT = {
				active_law:lawgroup_economic_system ?= {
					has_amendment = amendment_type:amendment_corporatized_economy
				}
			}
		}
		effect = {
			add_radicals = {
				value = 0.15
				pop_type = peasants
			}
			add_modifier = {
				name = modifier_corporatized_agriculture_failure
				days = 1825
			}
			ig:ig_landowners ?= {
				add_modifier = {
					name = modifier_ig_corporatized_agriculture_failure
					days = 1825
				}
			}
			ig:ig_rural_folk ?= {
				add_modifier = {
					name = modifier_ig_corporatized_agriculture_failure
					days = 1825
				}
			}
			remove_variable = corporatized_agriculture_months
			if = {
				limit = { has_variable = agrarian_subsidies_days_left }
				remove_variable = agrarian_subsidies_days_left
			}
			trigger_event = {
				id = corporatized_economy.13
				popup = yes
			}
		}
	}

	outcome = {
		header = JOURNAL_ENTRY_TIMEOUT
		effect_header = JOURNAL_ENTRY_ON_TIMEOUT
		flags = { timeout }
		conclusion_desc = JOURNAL_ENTRY_TIMEOUT_CONCLUSION_DESC
		effect = {
			add_radicals = {
				value = 0.15
				pop_type = peasants
			}
			add_modifier = {
				name = modifier_corporatized_agriculture_failure
				days = 1825
			}
			ig:ig_landowners ?= {
				add_modifier = {
					name = modifier_ig_corporatized_agriculture_failure
					days = 1825
				}
			}
			ig:ig_rural_folk ?= {
				add_modifier = {
					name = modifier_ig_corporatized_agriculture_failure
					days = 1825
				}
			}
			remove_variable = corporatized_agriculture_months
			if = {
				limit = { has_variable = agrarian_subsidies_days_left }
				remove_variable = agrarian_subsidies_days_left
			}
			trigger_event = {
				id = corporatized_economy.13
				popup = yes
			}
		}
	}

	current_value = {
		value = 0
		if = {
			limit = { has_variable = corporatized_agriculture_months }
			add = root.var:corporatized_agriculture_months
		}
	}

	goal_add_value = {
		add = 120
	}

	timeout = 3650

	progressbar = yes
	display_progressbar_as_months = yes

	weight = 100
}
