		ideology_national_socialist
		ideology_clerical_fascist
		ideology_corporatist
		ideology_integralist
	}

	# ... keep the rest of vanilla's movement_fascist blocks (creation_trigger, pop_support_trigger, etc.) intact ...

	character_support_trigger = {
		trigger_if = {
			limit = {
				is_in_exile_pool = yes
			}
			OR = {
				has_ideology = ideology:ideology_fascist
				has_ideology = ideology:ideology_national_socialist
				has_ideology = ideology:ideology_clerical_fascist
			}
		}
		trigger_else = {
			OR = {
				has_ideology = ideology:ideology_fascist
				has_ideology = ideology:ideology_national_socialist
				has_ideology = ideology:ideology_clerical_fascist
				has_ideology = ideology:ideology_corporatist
				has_ideology = ideology:ideology_integralist
			}
		}
	}

	character_support_weight = {
		value = 200
		if = {
			limit = {
				OR = {
					has_ideology = ideology:ideology_fascist
					has_ideology = ideology:ideology_national_socialist
					has_ideology = ideology:ideology_clerical_fascist
				}
			}
			multiply = {
				value = 5
			}
		}
	}
