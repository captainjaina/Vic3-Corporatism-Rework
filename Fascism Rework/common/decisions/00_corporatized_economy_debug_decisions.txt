debug_add_corporatized_economy_amendment = {
	is_shown = {
		is_ai = no
		NOT = {
			active_law:lawgroup_economic_system ?= {
				has_amendment = amendment_type:amendment_corporatized_economy
			}
		}
	}

	possible = {
		always = yes
	}

	when_taken = {
		if = {
			limit = {
				NOT = { has_technology_researched = corporatism }
			}
			add_technology_researched = corporatism
		}

		if = {
			limit = {
				NOT = { has_law = law_type:law_interventionism }
			}
			activate_law = law_type:law_interventionism
		}

		if = {
			limit = {
				any_interest_group = {
					is_in_government = yes
				}
			}
			random_interest_group = {
				limit = { is_in_government = yes }
				save_scope_as = debug_sponsor_ig
			}
		}
		else = {
			ig:ig_petty_bourgeoisie = {
				save_scope_as = debug_sponsor_ig
			}
		}

		active_law:lawgroup_economic_system = {
			add_amendment = {
				type = amendment_type:amendment_corporatized_economy
				sponsor = scope:debug_sponsor_ig
			}
		}
	}

	ai_chance = {
		value = 0
	}
}
