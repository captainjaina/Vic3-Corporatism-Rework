initiate_first_four_year_plan_decision = {
	is_shown = {
		NOT = { has_variable = first_four_year_plan_initiated }
		NOT = { has_journal_entry = je_first_four_year_plan }
		active_law:lawgroup_economic_system ?= {
			has_amendment = amendment_type:amendment_corporatized_economy
		}
	}

	possible = {
		active_law:lawgroup_economic_system ?= {
			has_amendment = amendment_type:amendment_corporatized_economy
		}
		OR = {
			has_law_or_variant = law_type:law_protectionism
			has_law_or_variant = law_type:law_isolationism
		}
	}

	when_taken = {
		set_variable = first_four_year_plan_initiated
		custom_tooltip = initiate_first_four_year_plan_decision_tt
	}

	ai_chance = {
		value = 10
	}
}

convene_planning_commission_decision = {
	is_shown = {
		has_variable = first_four_year_plan_completed
		NOT = { has_variable = fyp_active }
		NOT = { has_variable = fyp_selecting }
		active_law:lawgroup_economic_system ?= {
			has_amendment = amendment_type:amendment_corporatized_economy
		}
	}

	possible = {
		active_law:lawgroup_economic_system ?= {
			has_amendment = amendment_type:amendment_corporatized_economy
		}
		OR = {
			has_law_or_variant = law_type:law_protectionism
			has_law_or_variant = law_type:law_isolationism
		}
		NOT = { has_modifier = modifier_fyp_commission_review }
	}

	when_taken = {
		set_variable = fyp_selecting
		trigger_event = {
			id = corporatized_economy.5
			popup = yes
		}
		custom_tooltip = convene_planning_commission_decision_tt
	}

	ai_chance = {
		value = 10
	}
}
