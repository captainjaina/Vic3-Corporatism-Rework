# Enabling Act
corporatized_economy.14 = {
	type = country_event
	placement = root

	title = corporatized_economy.14.t
	desc = corporatized_economy.14.d
	flavor = corporatized_economy.14.f

	duration = 3

	event_image = {
		video = "unspecific_politicians_arguing"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/law_icons/single_party_state.dds"

	trigger = {
		NOT = { has_variable = enabling_act_event_fired }
		has_technology_researched = political_agitation
		NOR = {
			has_law = law_type:law_single_party_state
			has_law = law_type:law_autocracy
			has_law = law_type:law_theocracy
		}
		OR = {
			# Path 1: Influential governing Fascist ruler or Fascist-led governing IG
			ruler ?= {
				OR = {
					OR = {
						has_ideology = ideology:ideology_fascist
						has_ideology = ideology:ideology_national_socialist
						has_ideology = ideology:ideology_clerical_fascist
					}
					interest_group ?= {
						is_in_government = yes
						is_powerful = yes
						leader ?= {
							OR = {
								has_ideology = ideology:ideology_fascist
								has_ideology = ideology:ideology_national_socialist
								has_ideology = ideology:ideology_clerical_fascist
							}
						}
					}
				}
				interest_group ?= {
					is_in_government = yes
					is_powerful = yes
				}
			}
			# Path 2: Corporate State or Social Monarchy with an Authoritarian Petty Bourgeoisie ruler
			AND = {
				OR = {
					has_law = law_type:law_corporate_state
					has_law = law_type:law_social_monarchy
				}
				ruler ?= {
					has_ideology = ideology:ideology_authoritarian
					interest_group ?= {
						is_interest_group_type = ig_petty_bourgeoisie
						is_in_government = yes
					}
				}
			}
		}
	}

	immediate = {
		set_variable = enabling_act_event_fired
		ruler ?= {
			interest_group ?= {
				save_scope_as = fascist_enabling_ig
			}
			if = {
				limit = {
					OR = {
						has_ideology = ideology:ideology_fascist
						has_ideology = ideology:ideology_authoritarian
						has_ideology = ideology:ideology_national_socialist
						has_ideology = ideology:ideology_clerical_fascist
					}
				}
				save_scope_as = fascist_enabling_leader
			}
			else = {
				interest_group ?= {
					leader ?= {
						save_scope_as = fascist_enabling_leader
					}
				}
			}
		}
	}

	option = {
		name = corporatized_economy.14.a
		default_option = yes
		if = {
			limit = { has_law = law_type:law_monarchy }
			activate_law = law_type:law_social_monarchy
		}
		activate_law = law_type:law_single_party_state
		add_radicals = {
			value = 0.05
		}
		add_modifier = {
			name = modifier_enabling_act_emergency_powers
			days = 1825
		}
		every_interest_group = {
			limit = {
				is_in_government = no
			}
			add_modifier = {
				name = modifier_ig_enabling_act_opposition_outrage
				days = 1825
			}
		}
		ai_chance = {
			base = 75
		}
	}

	option = {
		name = corporatized_economy.14.b
		add_loyalists = {
			value = 0.10
		}
		every_interest_group = {
			limit = {
				is_in_government = no
			}
			add_modifier = {
				name = modifier_ig_enabling_act_constitutional_praise
				days = 1825
			}
		}
		every_interest_group = {
			limit = {
				is_in_government = yes
				OR = {
					leader ?= {
						has_ideology = ideology:ideology_fascist
						has_ideology = ideology:ideology_national_socialist
						has_ideology = ideology:ideology_clerical_fascist
					}
					any_scope_character = {
	this = root.ruler
	OR = {
		has_ideology = ideology:ideology_fascist
		has_ideology = ideology:ideology_authoritarian
		has_ideology = ideology:ideology_national_socialist
		has_ideology = ideology:ideology_clerical_fascist
	}
}
				}
			}
			add_modifier = {
				name = modifier_ig_enabling_act_fascist_betrayal
				days = 1825
			}
		}
		ai_chance = {
			base = 25
		}
	}
}

# Hidden Monthly Prerequisite Tracker for National Social Democracy JE
corporatized_economy.17 = {
	type = country_event
	hidden = yes

	immediate = {
		if = {
			limit = {
				NOT = { has_variable = national_social_democracy_completed }
				NOT = { has_variable = national_social_democracy_failed }
				OR = {
					has_law = law_type:law_corporate_state
					has_law = law_type:law_social_monarchy
				}
				OR = {
					has_law = law_type:law_census_voting
					has_law = law_type:law_universal_suffrage
				}
			}
			if = {
				limit = { has_variable = corporatist_social_democracy_prereq_months }
				change_variable = {
					name = corporatist_social_democracy_prereq_months
					add = 1
				}
			}
			else = {
				set_variable = {
					name = corporatist_social_democracy_prereq_months
					value = 1
				}
			}
		}
		else_if = {
			limit = { has_variable = corporatist_social_democracy_prereq_months }
			remove_variable = corporatist_social_democracy_prereq_months
		}
	}
}
