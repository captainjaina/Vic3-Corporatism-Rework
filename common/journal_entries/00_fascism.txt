# Adapted for all Fascist & Corporatist regimes
je_modernization_program = {
	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"

	group = je_group_internal_affairs

	activation_description = {
		always = yes
	}

	is_shown_when_inactive = {
		NOT = { has_variable = completed_modernization_program_var }
		has_technology_researched = political_agitation
		any_interest_group = {
			is_in_government = yes
			leader ?= {
				OR = {
					has_ideology = ideology:ideology_fascist
					has_ideology = ideology:ideology_national_socialist
					has_ideology = ideology:ideology_clerical_fascist
					has_ideology = ideology:ideology_integralist
					has_ideology = ideology:ideology_corporatist_leader
				}
			}
		}
	}

	possible = {
		OR = {
			has_law_or_variant = law_type:law_single_party_state
			has_law_or_variant = law_type:law_corporate_state
			has_law_or_variant = law_type:law_autocracy
			has_variable = white_terror_resolved
		}
		any_interest_group = {
			is_in_government = yes
			leader ?= {
				OR = {
					has_ideology = ideology:ideology_fascist
					has_ideology = ideology:ideology_national_socialist
					has_ideology = ideology:ideology_clerical_fascist
					has_ideology = ideology:ideology_integralist
					has_ideology = ideology:ideology_corporatist_leader
				}
			}
		}
	}

	immediate = {
		ordered_interest_group = {
			limit = {
				is_in_government = yes
				leader ?= {
					OR = {
						has_ideology = ideology:ideology_fascist
						has_ideology = ideology:ideology_national_socialist
						has_ideology = ideology:ideology_clerical_fascist
						has_ideology = ideology:ideology_integralist
						has_ideology = ideology:ideology_corporatist_leader
					}
				}
			}
			order_by = ig_clout
			position = 0
			check_range_bounds = no
			save_scope_as = fascist_ig
			set_variable = og_fascist_ig
			leader ?= {
				save_scope_as = fascist_leader
			}
		}
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			NOT = { has_law_or_variant = law_type:law_free_trade }
			custom_tooltip = {
				text = je_modernization_program_tt
				capital = {
					infrastructure >= 150
				}
			}
			any_scope_state = {
				filter = {
					is_incorporated = yes
				}
				any_scope_building = {
					is_building_type = building_railway
					level >= 2
				}
				percent = 1
			}
			any_scope_state = {
				is_incorporated = yes
				has_building = building_port
				count >= 2
			}
		}
		effect = {
			set_variable = {
				name = completed_modernization_program_var
			}
			trigger_event = { id = fascism_events.5 }
		}
	}

	on_monthly_pulse = {
	}

	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			NOT = {
				any_interest_group = {
					custom_tooltip = {
						text = fascism_1_tt
						leader ?= {
							OR = {
								has_ideology = ideology:ideology_fascist
								has_ideology = ideology:ideology_national_socialist
								has_ideology = ideology:ideology_clerical_fascist
								has_ideology = ideology:ideology_integralist
								has_ideology = ideology:ideology_corporatist_leader
							}
						}
					}
					is_in_government = yes
				}
			}
		}
		effect = {
			je_modernization_program_cleanup = yes
		}
	}

	weight = 10000
	can_deactivate = no

	should_be_pinned_by_default_uninvolved_or_context = yes
}

# Adapted to support National Socialism alongside Classical Fascism & Ethno-Nationalism
je_ethno_nationalist = {
	icon = "gfx/interface/icons/event_icons/event_military.dds"

	group = je_group_internal_affairs

	activation_description = {
		always = yes
	}

	is_shown_when_inactive = {
		NOT = { has_variable = completed_ethno_nationalism_var }
		any_interest_group = {
			any_scope_character = {
				OR = {
					AND = {
						prominence >= define:NCharacters|PROMINENCE_IMPORTANT_THRESHOLD
						has_role_of_type = politician
					}
					has_role_of_type = agitator
				}
				OR = {
					has_ideology = ideology:ideology_fascist
					has_ideology = ideology:ideology_national_socialist
					has_ideology = ideology:ideology_ethno_nationalist
				}
			}
		}
		NOT = { is_country_type = unrecognized }
		NOT = { is_country_type = decentralized }
	}

	possible = {
		any_interest_group = {
			is_in_government = yes
			any_scope_character = {
				OR = {
					AND = {
						prominence >= define:NCharacters|PROMINENCE_IMPORTANT_THRESHOLD
						has_role_of_type = politician
					}
					has_role_of_type = agitator
				}
				OR = {
					has_ideology = ideology:ideology_fascist
					has_ideology = ideology:ideology_national_socialist
					has_ideology = ideology:ideology_ethno_nationalist
				}
			}
		}
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			has_law_or_variant = law_type:law_ethnostate
			OR = {
				has_law_or_variant = law_type:law_migration_controls
				has_law_or_variant = law_type:law_closed_borders
			}
			custom_tooltip = {
				text = je_ethno_nationalist_tt
				NOT = {
					any_state = {
						is_homeland_of_country_cultures = ROOT
						NOT = { owner = ROOT }
					}
				}
			}
		}
		effect = {
			set_variable = {
				name = completed_ethno_nationalism_var
			}
			trigger_event = { id = fascism_events.6 }
		}
	}

	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			NOT = {
				any_interest_group = {
					custom_tooltip = {
						text = ethnonationalist_tt
						leader ?= {
							OR = {
								has_ideology = ideology:ideology_ethno_nationalist
								has_ideology = ideology:ideology_fascist
								has_ideology = ideology:ideology_national_socialist
							}
						}
					}
					is_in_government = yes
				}
			}
		}
		effect = {
			je_ethno_nationalist_cleanup = yes
		}
	}

	weight = 10000
	can_deactivate = no
}
