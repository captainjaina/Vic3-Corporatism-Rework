############################################################
# 1a. INDUSTRIAL PLAN: GENERAL ECONOMIC GROWTH
############################################################
je_fyp_general_industry = {
	icon = "gfx/interface/icons/law_icons/interventionism.dds"
	group = je_group_internal_affairs

	scripted_button = seize_idle_capital_button
	scripted_button = enforce_syndicate_quotas_button
	scripted_button = public_works_act_button
	scripted_button = enact_autarky_button

	activation_description = {
		has_variable = fyp_focus_general
		OR = {
			AND = {
				has_law = law_type:law_interventionism
				active_law:lawgroup_economic_system ?= {
					has_amendment = amendment_type:amendment_corporatized_economy
				}
			}
			AND = {
				has_law = law_type:law_command_economy
				OR = {
					has_law = law_type:law_corporate_state
					has_law = law_type:law_social_monarchy
				}
			}
		}
	}

	possible = {
		has_variable = fyp_focus_general
		OR = {
			AND = {
				has_law = law_type:law_interventionism
				active_law:lawgroup_economic_system ?= {
					has_amendment = amendment_type:amendment_corporatized_economy
				}
			}
			AND = {
				has_law = law_type:law_command_economy
				OR = {
					has_law = law_type:law_corporate_state
					has_law = law_type:law_social_monarchy
				}
			}
		}
	}

	immediate = {
		set_variable = { name = four_year_plan_months value = 0 }
		set_variable = {
			name = fyp_target_gdp
			value = {
				value = gdp
				multiply = 1.12
			}
		}
		set_variable = { name = fyp_current_metric value = 0 }
		every_scope_state = {
			every_scope_building = {
				limit = { is_building_type = building_railway }
				root = {
					change_variable = { name = fyp_current_metric add = prev.level }
				}
			}
		}
		if = {
			limit = { var:fyp_current_metric < 25 }
			set_variable = {
				name = fyp_target_railways
				value = {
					value = var:fyp_current_metric
					add = 5
				}
			}
		}
		else = {
			set_variable = {
				name = fyp_target_railways
				value = {
					value = var:fyp_current_metric
					multiply = 1.20
					round = yes
				}
			}
		}
		set_variable = {
			name = fyp_metric_surplus
			value = {
				value = var:fyp_current_metric
				subtract = var:fyp_target_railways
			}
		}
		set_variable = { name = fyp_current_light_levels value = 0 }
		every_scope_state = {
			every_scope_building = {
				limit = {
					OR = {
						is_building_type = building_food_industry
						is_building_type = building_textile_mill
						is_building_type = building_furniture_manufactory
						is_building_type = building_glassworks
						is_building_type = building_paper_mill
					}
				}
				root = {
					change_variable = { name = fyp_current_light_levels add = prev.level }
				}
			}
		}
		if = {
			limit = { var:fyp_current_light_levels < 4 }
			set_variable = {
				name = fyp_target_light_levels
				value = {
					value = var:fyp_current_light_levels
					add = 2
				}
			}
		}
		else = {
			set_variable = {
				name = fyp_target_light_levels
				value = {
					value = var:fyp_current_light_levels
					multiply = 1.30
					round = yes
				}
			}
		}
		set_variable = {
			name = fyp_light_surplus
			value = {
				value = var:fyp_current_light_levels
				subtract = var:fyp_target_light_levels
			}
		}
		trigger_event = { id = corporatized_economy.7 popup = yes }
	}

	on_monthly_pulse = {
		effect = {
			change_variable = { name = four_year_plan_months add = 1 }
			set_variable = { name = fyp_current_metric value = 0 }
			every_scope_state = {
				every_scope_building = {
					limit = { is_building_type = building_railway }
					root = {
						change_variable = { name = fyp_current_metric add = prev.level }
					}
				}
			}
			set_variable = {
				name = fyp_metric_surplus
				value = {
					value = var:fyp_current_metric
					subtract = var:fyp_target_railways
				}
			}
			set_variable = { name = fyp_current_light_levels value = 0 }
			every_scope_state = {
				every_scope_building = {
					limit = {
						OR = {
							is_building_type = building_food_industry
							is_building_type = building_textile_mill
							is_building_type = building_furniture_manufactory
							is_building_type = building_glassworks
							is_building_type = building_paper_mill
						}
					}
					root = {
						change_variable = { name = fyp_current_light_levels add = prev.level }
					}
				}
			}
			set_variable = {
				name = fyp_light_surplus
				value = {
					value = var:fyp_current_light_levels
					subtract = var:fyp_target_light_levels
				}
			}
		}
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			scope:journal_entry = { is_goal_complete = yes }
			custom_tooltip = {
				text = je_fyp_general_gdp_tt
				has_variable = fyp_target_gdp
				gdp >= root.var:fyp_target_gdp
			}
			custom_tooltip = {
				text = je_fyp_general_railways_tt
				has_variable = fyp_metric_surplus
				var:fyp_metric_surplus >= 0
			}
			custom_tooltip = {
				text = je_fyp_general_light_tt
				has_variable = fyp_light_surplus
				var:fyp_light_surplus >= 0
			}
		}
		effect = {
			add_modifier = { name = modifier_fyp_general_success days = 1825 }
			change_variable = { name = four_year_plan_concluded_count add = 1 }
			change_variable = { name = four_year_plan_success_count add = 1 }
			change_variable = { name = four_year_plan_number add = 1 }
			add_modifier = { name = modifier_fyp_commission_review days = 180 }
			if = {
				limit = {
					var:four_year_plan_success_count >= 5
					NOT = { has_variable = fyp_economic_miracle_achieved }
				}
				set_variable = fyp_economic_miracle_achieved
				trigger_event = { id = corporatized_economy.10 days = 2 popup = yes }
			}
			trigger_event = { id = corporatized_economy.8 popup = yes }
			remove_variable = fyp_active
			remove_variable = fyp_focus_general
			remove_variable = four_year_plan_months
			remove_variable = fyp_current_metric
			remove_variable = fyp_metric_surplus
			remove_variable = fyp_target_gdp
			remove_variable = fyp_target_railways
			remove_variable = fyp_current_light_levels
			remove_variable = fyp_target_light_levels
			remove_variable = fyp_light_surplus
		}
	}

	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			OR = {
				NOT = {
					OR = {
						AND = {
							has_law = law_type:law_interventionism
							active_law:lawgroup_economic_system ?= {
								has_amendment = amendment_type:amendment_corporatized_economy
							}
						}
						AND = {
							has_law = law_type:law_command_economy
							OR = {
								has_law = law_type:law_corporate_state
								has_law = law_type:law_social_monarchy
							}
						}
					}
				}
				NOR = {
					has_law_or_variant = law_type:law_protectionism
					has_law_or_variant = law_type:law_isolationism
				}
			}
		}
		effect = {
			add_radicals = { value = 0.05 }
			add_modifier = { name = modifier_fyp_general_failure days = 1825 }
			change_variable = { name = four_year_plan_concluded_count add = 1 }
			change_variable = { name = four_year_plan_number add = 1 }
			add_modifier = { name = modifier_fyp_commission_review days = 180 }
			trigger_event = { id = corporatized_economy.9 popup = yes }
			remove_variable = fyp_active
			remove_variable = fyp_focus_general
			remove_variable = four_year_plan_months
			remove_variable = fyp_current_metric
			remove_variable = fyp_metric_surplus
			remove_variable = fyp_target_gdp
			remove_variable = fyp_target_railways
			remove_variable = fyp_current_light_levels
			remove_variable = fyp_target_light_levels
			remove_variable = fyp_light_surplus
		}
	}

	outcome = {
		header = JOURNAL_ENTRY_TIMEOUT
		effect_header = JOURNAL_ENTRY_ON_TIMEOUT
		flags = { timeout }
		conclusion_desc = JOURNAL_ENTRY_TIMEOUT_CONCLUSION_DESC
		effect = {
			add_radicals = { value = 0.05 }
			add_modifier = { name = modifier_fyp_general_failure days = 1825 }
			change_variable = { name = four_year_plan_concluded_count add = 1 }
			change_variable = { name = four_year_plan_number add = 1 }
			add_modifier = { name = modifier_fyp_commission_review days = 180 }
			trigger_event = { id = corporatized_economy.9 popup = yes }
			remove_variable = fyp_active
			remove_variable = fyp_focus_general
			remove_variable = four_year_plan_months
			remove_variable = fyp_current_metric
			remove_variable = fyp_metric_surplus
			remove_variable = fyp_target_gdp
			remove_variable = fyp_target_railways
			remove_variable = fyp_current_light_levels
			remove_variable = fyp_target_light_levels
			remove_variable = fyp_light_surplus
		}
	}

	current_value = {
		value = 0
		if = {
			limit = { has_variable = four_year_plan_months }
			add = root.var:four_year_plan_months
		}
	}
	goal_add_value = { add = 48 }
	timeout = 1521
	progressbar = yes
	display_progressbar_as_months = yes
	weight = 100
}
