je_white_terror = {
	icon = "gfx/interface/icons/law_icons/secret_police.dds"

	group = je_group_crises

	scripted_button = je_white_terror_smash_underground
	scripted_button = je_white_terror_coopt_elites
	scripted_button = je_white_terror_subordinate_sword_altar
	scripted_button = je_white_terror_settle_accounts

	is_shown_when_inactive = {
		NOT = { has_variable = white_terror_resolved }
		OR = {
			has_law = law_type:law_single_party_state
			has_law = law_type:law_corporate_state
			has_law = law_type:law_autocracy
			has_variable = enabling_act_event_fired
		}
	}

	activation_description = {
		OR = {
			has_law = law_type:law_single_party_state
			has_law = law_type:law_corporate_state
			has_law = law_type:law_autocracy
			has_variable = enabling_act_event_fired
		}
		any_interest_group = {
			is_marginal = no
			leader ?= {
				NOR = {
					has_ideology = ideology:ideology_fascist
					has_ideology = ideology:ideology_national_socialist
					has_ideology = ideology:ideology_clerical_fascist
					has_ideology = ideology:ideology_corporatist_leader
					has_ideology = ideology:ideology_authoritarian
				}
			}
		}
	}

	possible = {
		OR = {
			has_law = law_type:law_single_party_state
			has_law = law_type:law_corporate_state
			has_law = law_type:law_autocracy
			has_variable = enabling_act_event_fired
		}
		any_interest_group = {
			is_marginal = no
			leader ?= {
				NOR = {
					has_ideology = ideology:ideology_fascist
					has_ideology = ideology:ideology_national_socialist
					has_ideology = ideology:ideology_clerical_fascist
					has_ideology = ideology:ideology_corporatist_leader
					has_ideology = ideology:ideology_authoritarian
				}
			}
		}
	}

	scripted_progress_bar = white_terror_progress_bar

# Outcome 1: Total State Consolidated (100 Progress)
	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			custom_tooltip = {
				text = je_white_terror_complete_bar_tt
				scope:journal_entry = {
					"scripted_bar_progress(white_terror_progress_bar)" >= 100
				}
			}
		}
		effect = {
			set_variable = {
				name = white_terror_resolved
				value = yes
			}
			if = {
				limit = {
					ruler ?= {
						OR = {
							has_ideology = ideology:ideology_fascist
							has_ideology = ideology:ideology_national_socialist
							has_ideology = ideology:ideology_clerical_fascist
						}
					}
				}
				ruler ?= { save_scope_as = regime_ideology_holder }
			}
			else = {
				ordered_interest_group = {
					limit = {
						is_in_government = yes
						leader ?= {
							OR = {
								has_ideology = ideology:ideology_fascist
								has_ideology = ideology:ideology_national_socialist
								has_ideology = ideology:ideology_clerical_fascist
							}
						}
					}
					order_by = ig_clout
					position = 0
					leader ?= { save_scope_as = regime_ideology_holder }
				}
			}

			if = {
				limit = {
					scope:regime_ideology_holder ?= { has_ideology = ideology:ideology_national_socialist }
				}
				add_modifier = {
					name = modifier_total_state_natsoc
					years = 10
				}
			}
			else_if = {
				limit = {
					scope:regime_ideology_holder ?= { has_ideology = ideology:ideology_clerical_fascist }
				}
				add_modifier = {
					name = modifier_total_state_clerical
					years = 10
				}
			}
			else = {
				add_modifier = {
					name = modifier_total_state_classical
					years = 10
				}
			}
		}
	}

	# Outcome 2: The Terror Backfires (0 Progress)
	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			custom_tooltip = {
				text = je_white_terror_fail_bar_tt
				scope:journal_entry = {
					"scripted_bar_progress(white_terror_progress_bar)" <= 0
				}
			}
		}
		effect = {
			set_variable = {
				name = white_terror_resolved
				value = yes
			}
			if = {
				limit = { has_modifier = modifier_mandate_of_the_march }
				remove_modifier = modifier_mandate_of_the_march
			}
			add_modifier = {
				name = modifier_white_terror_backfired
				years = 5
			}
			add_radicals = {
				value = 0.20
				pop_type = laborers
			}
			add_radicals = {
				value = 0.20
				pop_type = machinists
			}
			add_radicals = {
				value = 0.20
				pop_type = academics
			}
		}
	}

	weight = 100
}
