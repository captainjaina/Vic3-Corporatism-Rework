je_corporatized_economy = {
	icon = "gfx/interface/icons/law_icons/corporatized_unions.dds"

	group = je_group_internal_affairs

	is_shown_when_inactive = {
		has_law = law_type:law_interventionism
		NOT = {
			active_law:lawgroup_economic_system ?= {
				has_amendment = amendment_type:amendment_corporatized_economy
			}
		}
	}

	activation_description = {
		has_technology_researched = corporatism
		has_law = law_type:law_interventionism

		OR = {
			has_law = law_type:law_corporatized_unions
			has_law = law_type:law_corporate_state
			has_law = law_type:law_social_monarchy
		}

		OR = {
			custom_tooltip = {
				text = je_corporatized_economy_ruler_tt
				ruler = {
					OR = {
						has_ideology = ideology:ideology_fascist
						has_ideology = ideology:ideology_corporatist
						has_ideology = ideology:ideology_national_socialist
						has_ideology = ideology:ideology_clerical_fascist
					}
				}
			}
			any_scope_character = {
				interest_group = {
					is_in_government = yes
				}
				OR = {
					has_ideology = ideology:ideology_fascist
					has_ideology = ideology:ideology_corporatist
					has_ideology = ideology:ideology_national_socialist
						has_ideology = ideology:ideology_clerical_fascist
				}
			}
		}

		NOT = {
			active_law:lawgroup_economic_system ?= {
				has_amendment = amendment_type:amendment_corporatized_economy
			}
		}
	}

	possible = {
		has_technology_researched = corporatism
		has_law = law_type:law_interventionism

		OR = {
			has_law = law_type:law_corporatized_unions
			has_law = law_type:law_corporate_state
			has_law = law_type:law_social_monarchy
		}

		OR = {
			custom_tooltip = {
				text = je_corporatized_economy_ruler_tt
				ruler = {
					OR = {
						has_ideology = ideology:ideology_fascist
						has_ideology = ideology:ideology_corporatist
						has_ideology = ideology:ideology_national_socialist
						has_ideology = ideology:ideology_clerical_fascist
					}
				}
			}
			any_scope_character = {
				interest_group = {
					is_in_government = yes
				}
				OR = {
					has_ideology = ideology:ideology_fascist
					has_ideology = ideology:ideology_corporatist
					has_ideology = ideology:ideology_national_socialist
					has_ideology = ideology:ideology_clerical_fascist
				}
			}
		}

		NOT = {
			active_law:lawgroup_economic_system ?= {
				has_amendment = amendment_type:amendment_corporatized_economy
			}
		}
	}

	immediate = {
		if = {
			limit = {
				ruler = {
					OR = {
						has_ideology = ideology:ideology_fascist
						has_ideology = ideology:ideology_corporatist
						has_ideology = ideology:ideology_national_socialist
						has_ideology = ideology:ideology_clerical_fascist
					}
				}
			}
			ruler = {
				interest_group = {
					save_scope_as = corporatist_sponsor_ig
				}
			}
		}
		else = {
			random_scope_character = {
				limit = {
					interest_group = {
						is_in_government = yes
					}
					OR = {
						has_ideology = ideology:ideology_fascist
						has_ideology = ideology:ideology_corporatist
						has_ideology = ideology:ideology_national_socialist
						has_ideology = ideology:ideology_clerical_fascist
					}
				}
				interest_group = {
					save_scope_as = corporatist_sponsor_ig
				}
			}
		}

		active_law:lawgroup_economic_system = {
			add_amendment = {
				type = amendment_type:amendment_corporatized_economy
				sponsor = scope:corporatist_sponsor_ig
			}
		}
	}

	weight = 100
}
