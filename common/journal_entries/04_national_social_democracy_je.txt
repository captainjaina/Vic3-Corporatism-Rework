je_national_social_democracy = {
	icon = "gfx/interface/icons/law_icons/corporate_state.dds"

	group = je_group_internal_affairs

	is_shown_when_inactive = {
		NOT = { has_variable = national_social_democracy_completed }
		NOT = { has_variable = national_social_democracy_failed }
		OR = {
			has_law = law_type:law_corporate_state
			has_law = law_type:law_social_monarchy
		}
	}

	activation_description = {
		OR = {
			has_law = law_type:law_corporate_state
			has_law = law_type:law_social_monarchy
		}
		OR = {
			has_law = law_type:law_census_voting
			has_law = law_type:law_universal_suffrage
		}
		custom_tooltip = {
			text = je_national_social_democracy_5_years_tt
			has_variable = corporatist_social_democracy_prereq_months
			var:corporatist_social_democracy_prereq_months >= 60
		}
		has_technology_researched = human_rights
		has_technology_researched = feminism
		OR = {
			ruler ?= {
				OR = {
					has_ideology = ideology:ideology_fascist
					has_ideology = ideology:ideology_corporatist
					has_ideology = ideology:ideology_national_socialist
					has_ideology = ideology:ideology_clerical_fascist
				}
			}
			any_interest_group = {
				is_in_government = yes
				leader ?= {
					OR = {
						has_ideology = ideology:ideology_fascist
						has_ideology = ideology:ideology_corporatist
						has_ideology = ideology:ideology_national_socialist
						has_ideology = ideology:ideology_clerical_fascist
					}
				}
			}
		}
		average_sol >= 12
		any_scope_state = {
			filter = { is_incorporated = yes }
			percent >= 0.80
			turmoil < 0.25
		}
	}

	possible = {
		OR = {
			has_law = law_type:law_corporate_state
			has_law = law_type:law_social_monarchy
		}
		OR = {
			has_law = law_type:law_census_voting
			has_law = law_type:law_universal_suffrage
		}
		custom_tooltip = {
			text = je_national_social_democracy_5_years_tt
			has_variable = corporatist_social_democracy_prereq_months
			var:corporatist_social_democracy_prereq_months >= 60
		}
		has_technology_researched = human_rights
		has_technology_researched = feminism
		OR = {
			ruler ?= {
				OR = {
					has_ideology = ideology:ideology_fascist
					has_ideology = ideology:ideology_corporatist
					has_ideology = ideology:ideology_national_socialist
					has_ideology = ideology:ideology_clerical_fascist
				}
			}
			any_interest_group = {
				is_in_government = yes
				leader ?= {
					OR = {
						has_ideology = ideology:ideology_fascist
						has_ideology = ideology:ideology_corporatist
						has_ideology = ideology:ideology_national_socialist
						has_ideology = ideology:ideology_clerical_fascist
					}
				}
			}
		}
		average_sol >= 12
		any_scope_state = {
			filter = { is_incorporated = yes }
			percent >= 0.80
			turmoil < 0.25
		}
	}

	immediate = {
		set_variable = {
			name = national_social_democracy_months
			value = 0
		}
	}

	on_monthly_pulse = {
		effect = {
			change_variable = {
				name = national_social_democracy_months
				add = 1
			}
		}
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			has_law = law_type:law_worker_protections
			has_law = law_type:law_compulsory_primary_school
			OR = {
				has_law = law_type:law_wage_subsidies
				has_law = law_type:law_old_age_pension
			}
			has_law = law_type:law_slavery_banned
			has_law = law_type:law_public_schools
			has_law = law_type:law_public_health_insurance
			OR = {
				has_law = law_type:law_universal_suffrage
				has_law = law_type:law_women_in_the_workplace
				has_law = law_type:law_womens_suffrage
			}
			average_sol >= 18
			literacy_rate >= 0.75
		}
		effect = {
			set_variable = national_social_democracy_completed
			custom_tooltip = {
				text = je_national_social_democracy_convert_fascists_tt
				every_scope_character = {
					limit = {
						has_ideology = ideology:ideology_fascist
						has_ideology = ideology:ideology_national_socialist
						has_ideology = ideology:ideology_clerical_fascist
					}
					set_ideology = ideology:ideology_state_socialist
				}
			}
			add_modifier = {
				name = modifier_national_social_democracy
				days = 10950
			}
			ig:ig_trade_unions ?= {
				add_modifier = {
					name = modifier_ig_national_social_democracy_approval
					days = 10950
				}
			}
			ig:ig_petty_bourgeoisie ?= {
				add_modifier = {
					name = modifier_ig_national_social_democracy_approval
					days = 10950
				}
			}
			trigger_event = {
				id = corporatized_economy.16
				popup = yes
			}
			remove_variable = national_social_democracy_months
		}
	}

	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			NOR = {
				has_law = law_type:law_corporate_state
				has_law = law_type:law_social_monarchy
			}
		}
		effect = {
			set_variable = national_social_democracy_failed
			add_radicals = {
				value = 0.05
			}
			add_modifier = {
				name = modifier_national_social_democracy_failure
				days = 3650
			}
			every_political_movement = {
				add_modifier = {
					name = modifier_movement_national_social_democracy_failure
					days = 3650
				}
			}
			remove_variable = national_social_democracy_months
		}
	}

	outcome = {
		header = JOURNAL_ENTRY_TIMEOUT
		effect_header = JOURNAL_ENTRY_ON_TIMEOUT
		flags = { timeout }
		conclusion_desc = JOURNAL_ENTRY_TIMEOUT_CONCLUSION_DESC
		effect = {
			set_variable = national_social_democracy_failed
			add_radicals = {
				value = 0.05
			}
			add_modifier = {
				name = modifier_national_social_democracy_failure
				days = 3650
			}
			every_political_movement = {
				add_modifier = {
					name = modifier_movement_national_social_democracy_failure
					days = 3650
				}
			}
			remove_variable = national_social_democracy_months
		}
	}

	current_value = {
		value = 0
		if = {
			limit = { has_variable = national_social_democracy_months }
			add = root.var:national_social_democracy_months
		}
	}

	goal_add_value = {
		add = 240
	}

	timeout = 7305

	progressbar = yes
	display_progressbar_as_months = yes

	weight = 100
}
