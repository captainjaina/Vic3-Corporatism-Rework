namespace = rise_of_fascism

# 1. Climax Event: The March on the Capital (at 100 Fascist Momentum)
rise_of_fascism.1 = {
	type = country_event
	placement = root

	title = rise_of_fascism.1.t
	desc = rise_of_fascism.1.d
	flavor = rise_of_fascism.1.f

	duration = 3

	event_image = {
		video = "unspecific_military_parade"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	immediate = {
		random_interest_group = {
			limit = {
				leader ?= {
					OR = {
						has_ideology = ideology:ideology_fascist
						has_ideology = ideology:ideology_national_socialist
						has_ideology = ideology:ideology_clerical_fascist
					}
				}
			}
			save_scope_as = march_fascist_ig
			leader = {
				save_scope_as = march_fascist_leader
			}
		}
	}

	# Option A: Capitulate to the Ultimatum
	option = {
		name = rise_of_fascism.1.a
		default_option = yes

		if = {
			limit = { exists = scope:march_fascist_ig }
			scope:march_fascist_ig = {
				add_ruling_interest_group = yes
				add_modifier = {
					name = modifier_mandate_of_the_march_ig
					years = 4
				}
			}
		}
		if = {
			limit = {
				exists = scope:march_fascist_leader
				NOT = { has_law = law_type:law_monarchy }
				NOT = { has_law = law_type:law_social_monarchy }
			}
			scope:march_fascist_leader = {
				set_character_as_ruler = yes
			}
		}
		add_modifier = {
			name = modifier_mandate_of_the_march
			years = 4
		}
		add_radicals = {
			value = 0.10
			pop_type = laborers
		}
		add_radicals = {
			value = 0.15
			pop_type = academics
		}
	}

# Option B: Refuse the Ultimatum & Fight
	option = {
		name = rise_of_fascism.1.b

		add_radicals = {
			value = 0.25
			pop_type = shopkeepers
		}
		add_radicals = {
			value = 0.20
			pop_type = officers
		}
		add_radicals = {
			value = 0.15
			pop_type = clerks
		}
		every_interest_group = {
			limit = {
				leader ?= {
					OR = {
						has_ideology = ideology:ideology_fascist
						has_ideology = ideology:ideology_national_socialist
						has_ideology = ideology:ideology_clerical_fascist
					}
				}
			}
			add_modifier = {
				name = modifier_elites_resent_relief_spending
				years = 5
			}
		}
	}
	}

				name = modifier_elites_resent_relief_spending

# 4. Pulse Event: Demands for the Interior Ministry
rise_of_fascism.4 = {
	type = country_event
	placement = root

	title = rise_of_fascism.4.t
	desc = rise_of_fascism.4.d
	flavor = rise_of_fascism.4.f

	duration = 3

	event_image = {
		video = "unspecific_ruler_speaking_to_people"
	}

	trigger = {
		has_journal_entry = je_rise_of_fascism
		any_interest_group = {
			is_in_government = yes
			leader ?= {
				OR = {
					has_ideology = ideology:ideology_fascist
					has_ideology = ideology:ideology_national_socialist
					has_ideology = ideology:ideology_clerical_fascist
				}
			}
		}
		NOT = { has_variable = rise_of_fascism_ministry_cd }
	}

	immediate = {
		set_variable = {
			name = rise_of_fascism_ministry_cd
			days = 540
		}
	}

	# Refuse & Purge Fascist Ministers
	option = {
		name = rise_of_fascism.4.a
		default_option = yes

		je:je_rise_of_fascism ?= {
			add_progress = { value = -12 name = rise_of_fascism_progress_bar }
		}
		every_interest_group = {
			limit = {
				is_in_government = yes
				leader ?= {
					OR = {
						has_ideology = ideology:ideology_fascist
						has_ideology = ideology:ideology_national_socialist
						has_ideology = ideology:ideology_clerical_fascist
					}
				}
			}
			remove_ruling_interest_group = yes
			add_modifier = {
				name = modifier_elites_resent_relief_spending
				years = 3
			}
		}
	}

	# Hand Over the Ministries
	option = {
		name = rise_of_fascism.4.b

		je:je_rise_of_fascism ?= {
			add_progress = { value = 15 name = rise_of_fascism_progress_bar }
		}
		add_modifier = {
			name = modifier_deputized_order_squads
			years = 2
		}
	}
}
