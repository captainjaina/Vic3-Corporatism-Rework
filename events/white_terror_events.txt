# 2. The General Staff Demands Immunity
white_terror.2 = {
	type = country_event
	placement = root

	title = white_terror.2.t
	desc = white_terror.2.d
	flavor = white_terror.2.f

	duration = 3

	event_image = {
		video = "unspecific_military_parade"
	}

	trigger = {
		has_journal_entry = je_white_terror
		ig:ig_armed_forces ?= { is_marginal = no }
		NOT = { has_variable = white_terror_general_staff_cd }
		NOT = { has_variable = white_terror_subordinated_pillars }
	}

	immediate = {
		set_variable = {
			name = white_terror_general_staff_cd
			days = 730
		}
	}

	# Grant Military Autonomy
	option = {
		name = white_terror.2.a
		default_option = yes

		set_variable = { name = white_terror_general_staff_immune value = yes }
		add_modifier = {
			name = modifier_white_terror_general_staff_autonomy
			years = 5
		}
	}

	# Subordinate the General Staff
	option = {
		name = white_terror.2.b

		je:je_white_terror ?= {
			add_progress = { value = 8 name = white_terror_progress_bar }
		}
		add_modifier = {
			name = modifier_white_terror_purged_general_staff
			years = 3
		}
		if = {
			limit = { ruler ?= { has_ideology = ideology:ideology_national_socialist } }
			ig:ig_armed_forces ?= {
				leader ?= { set_ideology = ideology:ideology_national_socialist }
			}
		}
		else_if = {
			limit = { ruler ?= { has_ideology = ideology:ideology_clerical_fascist } }
			ig:ig_armed_forces ?= {
				leader ?= { set_ideology = ideology:ideology_clerical_fascist }
			}
		}
		else = {
			ig:ig_armed_forces ?= {
				leader ?= { set_ideology = ideology:ideology_fascist }
			}
		}
	}
}
